// Helper: check if a command-line argument exists
bool hasArg(int argc, char** argv, const char* arg) {
    for (int i = 1; i < argc; ++i) {
        if (std::strcmp(argv[i], arg) == 0) {
            return true;
        }
    }
    return false;
}

int main(int argc, char** argv) {
    hideCursor();
    
    // Parse command-line arguments
    bool isLoad = hasArg(argc, argv, "-load");
    bool isSave = hasArg(argc, argv, "-save");
    bool isSilent = hasArg(argc, argv, "-silent");
    
    // Create the appropriate Game subclass
    std::unique_ptr<Game> game;
    
    if (isLoad) {
        // Replay mode: load steps from file
        game = std::make_unique<GameFileInput>(isSilent);
    } else {
        // Normal mode: keyboard input
        game = std::make_unique<GameKeyboardInput>(isSave);
    }
    
    // Run the game
    game->run();
    
    // Print test result in silent mode
    if (isLoad && isSilent) {
        static_cast<GameFileInput*>(game.get())->printTestResult();
    }
    
    return 0;
}

================================================================================
                           HOW EACH MODE WORKS
================================================================================

NORMAL MODE (no arguments)
--------------------------
1. Menu appears
2. Player controls game with keyboard
3. No files created
4. Identical to Exercise 2 behavior

SAVE MODE (-save)
-----------------
1. Menu appears (same as normal)
2. Player controls game normally
3. All key presses recorded to Steps object
4. All game events recorded to Results object
5. On exit, files saved:
   - adv-world.steps - key presses with iteration numbers
   - adv-world.result - game events (deaths, screen changes, etc.)

LOAD MODE (-load)
-----------------
1. NO MENU - game starts immediately
2. Inputs come from adv-world.steps file
3. KEYBOARD IGNORED - you cannot control the game
4. Players move automatically (replaying the recording)
5. Speed is faster (10ms delay vs 50ms)
6. Game ends when replay completes

SILENT MODE (-load -silent)
---------------------------
1. NO VISUAL OUTPUT - screen stays blank
2. Runs as fast as possible (0ms delay)
3. Compares actual game events against adv-world.result
4. Prints "Test PASSED" or "Test FAILED: <reason>"
5. Used for automated testing/verification
